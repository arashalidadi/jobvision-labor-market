from scipy.optimize import linear_sum_assignment


def cluster_by_year(year, columns):
    frame = ads.loc[ads.jyear.eq(year) & ads.base_title.isin(eligible_titles)]
    counts = frame.groupby('base_title').size()
    keep = counts[counts.ge(30)].index
    skills_year = (skills_long.loc[skills_long.ad_id.isin(frame.ad_id) & skills_long.skill.isin(top_skills_list)]
                   .merge(ads[['ad_id', 'base_title']], on='ad_id')
                   .groupby(['base_title', 'skill']).ad_id.nunique().unstack(fill_value=0)
                   .reindex(columns=top_skills_list, fill_value=0))
    share = skills_year.div(counts, axis=0).fillna(0).reindex(keep)
    degree = (frame.assign(higher=frame.max_degree.isin(['کارشناسی ارشد', 'دکترا']))
              .groupby('base_title')[['higher']].mean().reindex(keep))
    experience = frame.groupby('base_title').Jobpost_RequiredExperienceYears.median().reindex(keep)
    matrix = share.join(degree.rename(columns={'higher': 'سهم مدرک ارشد و بالاتر'})).join(
        experience.rename('میانه سابقه')).fillna(0)
    return StandardScaler().fit_transform(matrix.values), matrix.index

        'مهارت متمایز دوم': f'{dominant.index[1]} ({dominant.iloc[1] * 100:+.1f} واحد)',
        'کمترین حضور': f'{weakest.index[0]} ({weakest.iloc[0] * 100:+.1f} واحد)',
        'میانه سابقه': block['میانه سابقه'].median(),

signature_table = pd.DataFrame(signature_rows).set_index('خوشه')
show_table('17.5', 'امضای خوشه‌ها: متمایزترین مهارت‌ها و عنوان‌های هسته هر خوشه', signature_table,
           formats={'تعداد عنوان': 'int', 'تعداد آگهی': 'int', 'میانه سابقه': 'hours',

تفسیر جدول ۱۷.۵: سه خوشه، سه بازار متمایز است:

  • خوشه ۱ — بدنهٔ عمومی بازار (۶۴۲ عنوان، ۲۳۳٬۶۳۸ آگهی، تنها ۲٪ مدرک ارشد): پرحجم‌ترین بخش بازار. عنوان‌های هسته «کارشناس فروش، حسابدار، کارشناس حسابداری» و مهارت متمایزش گرافیک و وب (+۰٫۶ واحد) است، در حالی که Excel در آن ۲٫۲ واحد کمتر از میانگین است. این خوشه جایی است که مهارت نرم‌افزاری تقاضای کمی دارد و نقش‌ها عمومی‌اند.
  • خوشه ۲ — مالی و حسابداری ارشد (۱۰۷ عنوان، ۱۹٬۴۱۳ آگهی، سابقهٔ میانه ۵ سال، ۱۵٪ ارشد): امضای آن Excel (+۱۳٫۵ واحد) و Word (+۹٫۲ واحد) است؛ عنوان‌های هسته «حسابدار ارشد، مدیر مالی، رئیس حسابداری». یعنی مسیر ارتقای مالی، برخلاف تصور، از مهارت‌های صفحه‌گسترده می‌گذرد نه از ابزارهای تخصصی.
  • خوشه ۳ — نقش‌های فنی-مقرراتی (۱۳ عنوان، ۱٬۴۰۰ آگهی، ۵۳٫۵٪ مدرک ارشد): کوچک‌ترین خوشه ولی مدرک‌محورترین بخش کل بازار (مسئول فنی، کارشناس رگولاتوری، نمایندگی فروش بیمه). اینجا مدرک ارشد عملاً شرط ورود است، نه مزیت.

جمع‌بندی خوشه‌بندی: مرز اصلی بازار «عمومی در برابر تخصصی» است، نه «فنی در برابر غیرفنی». خوشهٔ تخصصی حجم کمی دارد (۰٫۵٪ آگهی‌ها) ولی سخت‌ترین شرط‌های ورود را می‌گذارد؛ خوشهٔ مالی، سابقه‌محور است؛ و خوشهٔ عمومی، بزرگ‌ترین بازار برای جذب انبوه است. این سه، سه استراتژی متفاوت استخدام می‌خواهند.

پایداری خوشه‌ها با بازنمونه‌گیری: خوشه‌بندی یک بار روی داده انجام شده؛ پرسش درست این است که اگر ۲۰٪ عنوان‌ها را جابه‌جا کنیم، خوشه‌ها می‌مانند یا هر بار چیز دیگری درمی‌آید؟ معیار ARI (شاخص رند تعدیل‌شده) هر بار خوشه‌بندی بازنمونه را با خوشه‌بندی پایه مقایسه می‌کند؛ بالای ۰٫۷ یعنی ساختار پایدار.

# ۱۷.۶ — پایداری خوشه‌ها با بازنمونه‌گیری
baseline_labels = title_features_clustered['خوشه'].to_numpy()
stability_rng = np.random.default_rng(RANDOM_STATE)
stability_scores = []
title_count = len(baseline_labels)
for _ in range(25):
    chosen = stability_rng.choice(title_count, size=int(title_count * 0.8), replace=False)
    labels = KMeans(n_clusters=best_k, n_init=10, random_state=RANDOM_STATE).fit_predict(X_title[chosen])
    stability_scores.append(adjusted_rand_score(baseline_labels[chosen], labels))

stability_scores = np.array(stability_scores)
stability_ari_table = pd.DataFrame({
    'مقدار': [float(np.mean(stability_scores)), float(np.std(stability_scores)),
              float(np.percentile(stability_scores, 5)), float(np.min(stability_scores)),
              float((stability_scores >= .7).mean() * 100)],
}, index=['میانگین ARI', 'انحراف معیار', 'صدک پنجم', 'کمینه', 'سهم تکرارهای پایدار (٪)'])

           formats={'مقدار': 'score'}, sample=f'{title_count} عنوان شغلی، {best_k} خوشه',
