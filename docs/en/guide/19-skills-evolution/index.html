Interpretation of Table 19.1 and Figure 19.1: The market’s skill portfolio has changed slowly but directionally over two years. The largest increases: WordPress +1.4 percentage points, Microsoft Word +1.1, Google Analytics +1.0, Sepidar +0.8, and Excel +0.7. The largest declines: Microsoft Outlook −4.2, PowerPoint −4.0, Microsoft Dynamics −1.1, Microsoft Project −1.0, HTML/CSS −0.8, and JavaScript −0.7.

Two narratives emerge from these numbers. First, a shift toward digital marketing: WordPress, Google Analytics, and Keyword Planner have grown, consistent with the growth in the share of digital marketing in Section 10. Second, the replacement of heavyweight office tools: Outlook, PowerPoint, and Project have given way to lighter and more communicative tools; part of this decline may stem from a change in how skills are recorded in job postings, not a real decline in need.

Warning: Programming skills (JavaScript, HTML/CSS, C#) have declined in their share, at the same time as the decline in the share of the programming category. These two data points confirm each other.

Interpretation of Table 19.2 and Figure 19.2: With the condition of at least 50 job postings in both years (to remove explosive small-sample growth), the fastest growth belongs to barista (+422%), cook (+318%), Instagram admin (+309%), packaging worker (+295%), website designer (+264%), and safety officer (+259%). title_skill_share = (skills_long.merge(ads[['ad_id', 'base_title', 'jyear']], on='ad_id')
                     .groupby(['base_title', 'jyear', 'skill']).ad_id.nunique().unstack(fill_value=0))
title_ads_year = (ads.assign(has_skill=ads.ad_id.isin(skills_long.ad_id))
                  .groupby(['base_title', 'jyear']).has_skill.sum())
title_skill_rate = title_skill_share.div(title_ads_year, axis=0) * 100

stability_rows = []
top_titles_by_volume = ads.base_title.value_counts().head(12).index
for title in top_titles_by_volume:
    try:
        first = set(title_skill_rate.loc[(title, 1401)].dropna().loc[lambda s: s.ge(10)].index)
        second = set(title_skill_rate.loc[(title, 1402)].dropna().loc[lambda s: s.ge(10)].index)
    except KeyError:
        continue
    if not first or not second:
        continue
    jaccard = len(first & second) / len(first | second)
    joined = pd.DataFrame({'۱۴۰۱': title_skill_rate.loc[(title, 1401)],
                           '۱۴۰۲': title_skill_rate.loc[(title, 1402)]}).dropna()

        'عنوان شغلی': title, 'تعداد آگهی': int(ads.base_title.eq(title).sum()),
        'شباهت سبد مهارت': jaccard,
        'مهارت در حال رشد': joined['تغییر'].idxmax() if len(joined) else np.nan,

19.4. Net premium of skills with controls #

Section 22.4 measured the premium of each skill within “its own job category” and descriptively. Here, with a ridge model, the premium of all skills simultaneously is estimated while controlling for work experience, hours, company size and age, degree level, and number of benefits; that is, the contribution of each skill after subtracting the effect of the others. This table is a direct input to the skill bundle.

Show the code for this step52 lines
Python
# ۱۹.۴ — پاداش خالص مهارت با کنترل
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline

skill_premium_frame = paid.copy()
skill_premium_frame['exp_years'] = skill_premium_frame.Jobpost_RequiredExperienceYears.fillna(0)
skill_premium_frame['hours'] = skill_premium_frame.weekly_hours.fillna(skill_premium_frame.weekly_hours.median())
skill_premium_frame['size_rank'] = skill_premium_frame.company_size_rank.fillna(skill_premium_frame.company_size_rank.median())
skill_premium_frame['age_rank'] = skill_premium_frame.company_age_rank.fillna(skill_premium_frame.company_age_rank.median())
skill_premium_frame['degree_rank'] = skill_premium_frame.max_degree_rank.fillna(0)

ext_skill_cols = list(skills_long.skill.value_counts().head(20).index)
skill_design = (skills_long.loc[skills_long.skill.isin(ext_skill_cols)]
                .assign(flag=1)
                .pivot_table(index='ad_id', columns='skill', values='flag', fill_value=0)
                .reindex(skill_premium_frame.ad_id, fill_value=0))
skill_premium_frame = pd.concat([skill_premium_frame.reset_index(drop=True),
                                 skill_design.reset_index(drop=True)], axis=1)

skill_numeric = ['exp_years', 'hours', 'size_rank', 'age_rank', 'degree_rank', 'benefit_count']
skill_features = skill_numeric + ext_skill_cols
skill_model = make_pipeline(StandardScaler(), Ridge(alpha=1.0))
skill_model.fit(skill_premium_frame[skill_features], skill_premium_frame.log_salary)
skill_coef = pd.Series(skill_model.named_steps['ridge'].coef_, index=skill_features)

skill_premium_table = pd.DataFrame({
    'تعداد آگهی': [int(skill_design[column].sum()) for column in ext_skill_cols],
    'پاداش خالص (٪)': [(math.exp(skill_coef[column]) - 1) * 100 for column in ext_skill_cols],
}, index=ext_skill_cols).sort_values('پاداش خالص (٪)', ascending=False)
show_table('19.4', 'پاداش خالص بیست مهارت پرتکرار (با کنترل هم‌زمان سایر ویژگی‌ها)', skill_premium_table,
           formats={'تعداد آگهی': 'int', 'پاداش خالص (٪)': 'pp'},
           sample=f'{len(skill_premium_frame):,} آگهی دارای حقوق', index_name='مهارت',
           note='اثر خالص هر مهارت پس از کنترل سابقه، ساعت، اندازه و قدمت شرکت، مقطع و شمار مزایا.')

fig, ax = plt.subplots(figsize=(10.6, 6.4))
skill_plot = skill_premium_table.sort_values('پاداش خالص (٪)')
colors = [RED if value < 0 else TEAL for value in skill_plot['پاداش خالص (٪)']]
ax.barh([fa(name) for name in skill_plot.index], skill_plot['پاداش خالص (٪)'], color=colors, height=.72)
ax.axvline(0, color=DARK, lw=1)
bar_labels(ax, fmt='{:+.1f}', horizontal=True, fontsize=8)
heading(ax, '۱۹.۳', 'پاداش خالص مهارت‌ها پس از کنترل سایر ویژگی‌ها')
axes_labels(ax, 'پاداش خالص (٪ نسبت به میانگین همان مهارت‌داران)', 'مهارت')
complete('19.3')

best_skill = skill_premium_table.index[0]
worst_skill = skill_premium_table.index[-1]
show_metrics('پاداش مهارت', [
    ('بیشترین پاداش', f'{best_skill} — {pf(skill_premium_table["پاداش خالص (٪)"].iloc[0])}'),

Key numbers: skill premium

  • Highest premium: GIT — 2.6%
  • Lowest premium: Adobe Photoshop — -2.5%

Interpretation of Table 19.4 and Figure 19.3: The net premium of skills shows that the Iranian market pays for specialized tools, not for general tools:

  • At the top: GIT +2.6%, WordPress +1.5%, SQL Server +1.4%, JavaScript +1.1%, Microsoft Project +0.8%, AutoCAD +0.7%.
  • At the bottom: Adobe Photoshop −2.5%, Microsoft Word −2.4%, Excel −0.6%, Outlook −0.5% (10 of 20 skills are positive, meaning there is a real distinction).

Why are general tools negative? Because in this model job category is not controlled, and Word/Excel/Photoshop appear more in administrative, service, and entry-level job postings; so their negative coefficient does not indicate that the skill is “low-value” — it indicates an association with low-salary roles. The comparison with Section 22.4 (the premium of a skill within its own category) completes this distinction: there, general tools become neutral and specialized skills remain positive.

A practical takeaway for writing a job posting: Writing “proficient in Word and Excel” creates no differentiation; it is mentioning the specialized tools of the same role (for example, Git for a programmer, SQL Server for an analyst, AutoCAD for an engineer) that sets a posting apart from competitors.

19.5. The words inside a job title: what is each word worth? #

Job titles in Iran are built from a handful of recurring words: specialist, manager, operator, officer, assistant… Separating those words lets us estimate what salary level each one signals — without a model. This is a simple version of title topic mining: instead of character-level TF-IDF it uses real tokens, because the Persian zero-width non-joiner breaks word boundaries and character n-grams produce unreadable output.

Python
# ۱۹.۵ — واژه‌های سازندهٔ عنوان شغلی و سطح حقوقی هر واژه
def title_tokens(text):
    cleaned = str(text).replace('\u200c', ' ').replace('-', ' ')
    return [token for token in cleaned.split() if len(token) > 2]


token_rows = []
for ad_id, title in zip(ads.ad_id, ads.base_title):
    for token in set(title_tokens(title)):
        token_rows.append((ad_id, token))
token_long = pd.DataFrame(token_rows, columns=['ad_id', 'token'])

token_stats = (token_long.groupby('token')
               .agg(**{'تعداد آگهی': ('ad_id', 'nunique')})
               .query('`تعداد آگهی` >= 1500'))
token_stats = token_stats.join(ads.groupby('ad_id').salary_mid.median().rename('حقوق'),
                               how='left')
paid_by_ad = paid.groupby('ad_id').salary_mid.median()
token_stats['میانه حقوق'] = token_stats.index.map(
    lambda token: paid.loc[paid.ad_id.isin(token_long.loc[token_long.token.eq(token), 'ad_id']), 'salary_mid'].median())
token_stats = token_stats.drop(columns=['حقوق'])
token_stats['سهم آگهی'] = token_stats['تعداد آگهی'] / len(ads) * 100
token_stats = token_stats.sort_values('میانه حقوق', ascending=False)

keyword_table = token_stats.head(18).copy()
show_table('19.5', 'میانه حقوق آگهی‌هایی که این واژه در عنوانشان آمده', keyword_table,
           formats={'تعداد آگهی': 'int', 'سهم آگهی': 'pct', 'میانه حقوق': 'money'},
           sample='واژه‌های با دست‌کم ۱٬۵۰۰ آگهی، مرتب بر پایه میانه حقوق', index_name='واژه')

fig, ax = plt.subplots(figsize=(10.4, 6.2))
plot_tokens = keyword_table.sort_values('میانه حقوق')
colors = [TEAL if value >= paid.salary_mid.median() else RED for value in plot_tokens['میانه حقوق']]
ax.barh([fa(token) for token in plot_tokens.index], plot_tokens['میانه حقوق'], color=colors, height=.72)
ax.axvline(paid.salary_mid.median(), color=DARK, ls='--', lw=1.2)
bar_labels(ax, fmt='{:,.1f}', horizontal=True, fontsize=8)
heading(ax, '۱۹.۴', 'میانه حقوق بر پایه واژه‌های سازندهٔ عنوان شغلی')
axes_labels(ax, 'میانه حقوق (میلیون تومان)', 'واژه در عنوان شغلی')
complete('19.4')

show_metrics('۱۹.۳', [
    ('بالاترین واژه', f'{keyword_table.index[0]} — {nf(keyword_table["میانه حقوق"].iloc[0], 1)} میلیون'),
    ('پایین‌ترین واژه', f'{keyword_table.index[-1]} — {nf(keyword_table["میانه حقوق"].iloc[-1], 1)} میلیون'),
    ('شمار واژه‌های بررسی‌شده', nf(len(token_stats))),
])
