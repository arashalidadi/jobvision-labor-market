Reading tables 6.1 and 6.2: the market aggregates into three wage layers: 16.8% of postings below 8 million, 71.9% between 8 and 20 million and 11.3% above 20 million tomans. The middle layer is the body of the administrative, sales and services labor market; the layer above 20 million, which accounts for less than one ninth of postings, belongs to specialized, engineering and managerial roles.

The type of declared range also reveals a managerial pattern: no posting declared a range narrower than 15%; that is, Iranian employers almost always build a negotiation margin of at least 15 percent into the posting. This is itself a practical recommendation: if you want the posting to look more professional, declare a range of about 20 to 25 percent — not a narrow range that only shows the expectation ceiling, and not a very wide range that leads to the posting losing credibility.

6.2. Extracting working hours from free text #

The “working hours” text is almost unique for every posting (68,247 different cases). To turn it into a number, we write a rule-based parser that detects hour ranges, day-of-week ranges, daily hours and explicit weekly hours, and labels the result with its extraction source.

FA_DIGITS_MAP = str.maketrans('۰۱۲۳۴۵۶۷۸۹٠١٢٣٤٥٦٧٨٩', '01234567890123456789')
DAY_FIXES = [('یک شنبه', 'یکشنبه'), ('دو شنبه', 'دوشنبه'), ('سه شنبه', 'سه‌شنبه'),
             ('چهار شنبه', 'چهارشنبه'), ('پنج شنبه', 'پنجشنبه'), ('پنج‌شنبه', 'پنجشنبه')]
WEEK_DAYS = [('یکشنبه', 1), ('دوشنبه', 2), ('سه‌شنبه', 3), ('چهارشنبه', 4),
             ('پنجشنبه', 5), ('جمعه', 6), ('شنبه', 0)]
DAY_ORDER = dict(WEEK_DAYS)
DAY_RE = '|'.join(re.escape(name) for name, _ in WEEK_DAYS)
TIME_RE = re.compile(r'(\d{1,2})\s*(?:[:.]\s*(\d{2}))?\s*(?:الی|تا|-|–|به)\s*(\d{1,2})\s*(?:[:.]\s*(\d{2}))?')
DAYRANGE_RE = re.compile(rf'({DAY_RE})\s*(?:تا|الی|-|–|لغایت)\s*({DAY_RE})')


def normalize_worktime(value):
    text = str(value).translate(FA_DIGITS_MAP).replace('\u200c', ' ').replace('\xa0', ' ')
    text = re.sub(r'\s+', ' ', text)
    for bad, good in DAY_FIXES:
        text = text.replace(bad, good)
    return text.strip()


def clock(value, minutes):
    hour = value + minutes / 60
    return hour + 12 if hour < 6 else hour


def valid_span(start, end):
    if end <= start:
        end += 12
    span = end - start
    return (start, end, span) if 4 <= span <= 14 else None


def segment_days(segment):
    """تعداد روزهای هفته از یک قطعه متن: بازه روزها، فهرست روزها یا «N روز در هفته»."""
    day_range = DAYRANGE_RE.search(segment)
    if day_range:
        start, end = DAY_ORDER[day_range.group(1)], DAY_ORDER[day_range.group(2)]
        return (end - start) % 7 + 1
    found = {DAY_ORDER[m.group(0)] for m in re.finditer(DAY_RE, segment)}
    if found:
        return len(found)
    count = re.search(r'(\d{1,2})\s*روز\s*(?:کاری|در هفته|از هفته|هفته)', segment)
    if count and 1 <= int(count.group(1)) <= 7:
        return int(count.group(1))
    return None


def parse_worktime(value, work_type):
    """تجزیه متن ساعات کاری به ساعت هفتگی، روزهای هفته و پرچم‌های شیفت."""
    text = normalize_worktime(value)
    weekly = np.nan
    source = 'نامعلوم'
    daily_default = 5.5 if work_type == 'تمام وقت' else 3.0

    matches = list(TIME_RE.finditer(text))
    pieces, ranges = [], []
    for order, match in enumerate(matches):
        start = clock(int(match.group(1)), int(match.group(2) or 0))
        end = clock(int(match.group(3)), int(match.group(4) or 0))
        cleaned = valid_span(start, end)
        if cleaned is None:
            continue
        start, end, span = cleaned
        boundary = matches[order - 1].end() if order else 0
        pieces.append((span, segment_days(text[boundary:match.start()])))
        ranges.append((start, end))

    week_match = re.search(r'(\d{1,3})\s*ساعت\s*(?:در|از)\s*هفته', text)
    month_match = re.search(r'(\d{1,3})\s*ساعت\s*(?:در|از)\s*ماه', text)
    daily_match = re.search(r'(?:روزی|هر روز|در روز)\s*(\d{1,2})(?:[.,](\d))?\s*ساعت', text)

    if pieces and any(days for _, days in pieces):
        full = [span * days for span, days in pieces if days]
        partial = [span * daily_default for span, days in pieces if not days]
        weekly = sum(full) + sum(partial)
        source = 'بازه و روز از متن' if not partial else 'بازه و روز از متن (روز پیش‌فرض برای بخشی)'
    elif week_match:
        weekly, source = float(week_match.group(1)), 'صریح هفتگی'
    elif month_match:
        weekly, source = float(month_match.group(1)) / (52 / 12), 'صریح ماهانه'
    elif daily_match:
        daily = float(daily_match.group(1)) + (float('0.' + daily_match.group(2)) if daily_match.group(2) else 0)
        weekly, source = daily * (segment_days(text) or daily_default), 'ساعت روزانه × روز هفته'
    elif pieces:
        weekly, source = sum(span for span, _ in pieces) * daily_default, 'بازه با روز پیش‌فرض'
    else:
        days = segment_days(text)
        if days:
            weekly, source = 8 * days, 'روز بدون ساعت'
        else:
            defaults = {'تمام وقت': 44.0, 'تمام وقت یا پاره وقت': 44.0, 'پاره وقت': 22.0}
            weekly, source = defaults.get(work_type, np.nan), 'پیش‌فرض نوع همکاری'

    if not np.isnan(weekly) and not 8 <= weekly <= 84:
        weekly, source = np.nan, 'نامعتبر (خارج از بازه معقول)'

    return {
        'weekly_hours': weekly,
        'days_week': segment_days(text),
        'start_hour': ranges[0][0] if ranges else np.nan,
        'end_hour': ranges[0][1] if ranges else np.nan,
        'night_shift': bool(re.search(r'شبانه|شیفت شب|شب کاری|شب‌کاری', text)),
        'flexible_hours': bool(re.search(r'منعطف|شناور', text)),
        'has_thursday': 'پنجشنبه' in text,
        'has_friday': 'جمعه' in text,
        'hours_source': source,
    }

Reading tables 6.3 and 6.4 and figures 6.1 and 6.2: weekly hours were estimated for 97.6% of postings. The extraction source matters: 59.2% came from the free text (hour range × days of the week), which is the most accurate case; 17.6% had only the days of the week and were estimated assuming 8 hours per day; 13.7% had no hourly information and were filled based on work type (44 hours for full-time and 22 hours for part-time). 1.3% also had an invalid value and were left empty.

The median weekly hours is 45 hours and the interquartile range is 43 to 48 hours; slightly above the legal standard of 44 hours, because some postings are six-day and some have several time blocks within one day. The working-week pattern is also entirely Iranian: 43.5% of postings listed Thursday among the working days, while Friday is a working day in only 1.5% of postings.

Note that this estimate is “composite” in nature. In later sections, each hourly income analysis is performed on the same subset that has valid hours, and the limitation of this estimate is reported explicitly.

def load_json_list(value):
    if not isinstance(value, str) or not value.strip():
        return []
    try:
        data = json.loads(value)
    except (ValueError, TypeError):
        return []
    return data if isinstance(data, list) else []

Interpretation of tables 6.5 and 6.6 and figure 6.3: 193,742 job postings (53.8%) have a skills list, and altogether 491 distinct skills are requested in them; on average each job posting asks for 3 skills, and at most 69.

The most frequent list gives a picture of the nature of Iran's labor market: the top four skills all come from the Microsoft Office suite (Excel 38.4%, Word 33.8%, PowerPoint 12.8%, and Outlook 10.9% of all job postings). This means that “digital office literacy” is the most in-demand skill in the market, not programming. After that, the second category belongs to Iranian accounting and administrative software (Sepidar, Rahkaran, Hamkaran System, along with Access and Project), which is a local cluster rarely seen in global analyses. The third category is design and engineering tools (Photoshop, Illustrator, AutoCAD, SolidWorks, CATIA), and programming and data skills (SQL Server, HTML/CSS, Git, JavaScript, Python) come in later ranks.

A managerial point: the median salary of job postings with a skills list is 13.5 million, versus 11 million for job postings without skills. This difference shows that “stating skills” is itself a sign that the role is structured and formal, not merely a technical list; in section 22 the premium of each skill is measured while controlling for job category.

Interpretation of tables 6.7 and 6.8: Only 47.1% of job postings specify a degree level, and in more than half of job postings this requirement is not stated. Among the job postings that do state it, a bachelor's degree dominates with 38.9% of all job postings; an associate degree 4.3%, a master's degree 3.5%, and a doctorate 0.4%. That is, Iran's labor market is in practice a “bachelor's market,” and postgraduate education is a condition of a small minority of job postings.

The pay pattern is clear: the median salary of bachelor's job postings is 13.5 million and of master's and higher job postings 17.5 million tomans; a gap of nearly 30%. TITLE_NOISE = r'\b(خانم|آقا|فوری|تمام وقت|پاره وقت|نیمه وقت|دورکاری|حضوری|آنلاین)\b'


def base_title(title):
    """عنوان شغلی پایه: حذف توضیحات پرانتزی، پسوندهای جنسیت و بخش‌های فرعی."""
    text = clean_text(title)
    text = re.sub(r'\(.*?\)', ' ', text)
    text = re.split(r'[-–—|/,،:؛]', text)[0]
    text = re.sub(TITLE_NOISE, ' ', text, flags=re.IGNORECASE)
    text = re.sub(r'\s+', ' ', text).strip(' .')
    return text or clean_text(title)


def seniority_level(title):
    text = title.lower()
    if re.search(r'کارآموز|intern|trainee', text):
        return 'کارآموز'
    if re.search(r'مدیر|manager|head of|ریاست', text):
        return 'مدیریتی'

        return 'سرپرستی'
    if re.search(r'ارشد|senior|expert senior', text):
        return 'کارشناسی ارشد'
    if re.search(r'کارشناس|expert|specialist', text):
        return 'کارشناسی'
    return 'سایر'

def ordinal_pairs(column, rank_column):
    pairs = (ads[[column, rank_column]].dropna().drop_duplicates().sort_values(rank_column))
    return '؛ '.join(f'{value} ← {int(rank)}' for value, rank in pairs.itertuples(index=False))


def levels_text(column, limit=6):
    counts = ads[column].fillna('نامشخص').value_counts()
    head = '، '.join(str(name) for name in counts.head(limit).index)
    rest = counts.size - min(limit, counts.size)
    return head + (f' … (+{rest} سطح دیگر)' if rest > 0 else '')
